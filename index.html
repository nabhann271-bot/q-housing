from flask import Flask, render_template, request, redirect, url_for, jsonify from flask_sqlalchemy import SQLAlchemy from datetime import datetime app = Flask(__name__) app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///employees.db' app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False db = SQLAlchemy(app) # نموذج الموظف class Employee(db.Model): id = db.Column(db.Integer, primary_key=True) name = db.Column(db.String(100), nullable=False) national_id = db.Column(db.String(20), unique=True, nullable=False) room_number = db.Column(db.String(10), nullable=False) phone = db.Column(db.String(15)) check_in = db.Column(db.DateTime, default=datetime.utcnow) status = db.Column(db.String(20), default='مقيم') # مقيم، مغادر def __repr__(self): return f'' # نموذج الغرفة class Room(db.Model): id = db.Column(db.Integer, primary_key=True) room_number = db.Column(db.String(10), unique=True, nullable=False) capacity = db.Column(db.Integer, default=2) current_occupants = db.Column(db.Integer, default=0) # الصفحة الرئيسية @app.route('/') def index(): employees = Employee.query.all() rooms = Room.query.all() return render_template('index.html', employees=employees, rooms=rooms) # إضافة موظف @app.route('/add_employee', methods=['POST']) def add_employee(): name = request.form['name'] national_id = request.form['national_id'] room_number = request.form['room_number'] phone = request.form.get('phone', '') # التحقق من وجود الغرفة room = Room.query.filter_by(room_number=room_number).first() if not room: # إنشاء الغرفة إذا لم توجد room = Room(room_number=room_number, capacity=2, current_occupants=0) db.session.add(room) # التحقق من السعة if room.current_occupants >= room.capacity: return "الغرفة ممتلئة!", 400 employee = Employee( name=name, national_id=national_id, room_number=room_number, phone=phone ) room.current_occupants += 1 db.session.add(employee) db.session.commit() return redirect(url_for('index')) # تسجيل خروج موظف @app.route('/checkout/') def checkout(id): employee = Employee.query.get_or_404(id) room = Room.query.filter_by(room_number=employee.room_number).first() if room: room.current_occupants -= 1 employee.status = 'مغادر' db.session.commit() return redirect(url_for('index')) # حذف موظف @app.route('/delete/') def delete(id): employee = Employee.query.get_or_404(id) db.session.delete(employee) db.session.commit() return redirect(url_for('index')) # API - جلب بيانات موظف (للواجهة) @app.route('/api/employees') def api_employees(): employees = Employee.query.all() return jsonify([{ 'id': e.id, 'name': e.name, 'national_id': e.national_id, 'room': e.room_number, 'phone': e.phone, 'status': e.status } for e in employees]) if __name__ == '__main__': with app.app_context(): db.create_all() # إضافة غرف افتراضية إذا لم توجد if Room.query.count() == 0: for i in range(1, 11): db.session.add(Room(room_number=f'غرفة {i}', capacity=2)) db.session.commit() app.run(debug=True, host='0.0.0.0', port=5000)
